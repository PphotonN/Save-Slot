static void appendUtf8(char16_t*& out, char16_t* end, const char* s)
{
    const unsigned char* p=(const unsigned char*)s;
    while (*p && out+1<end)
    {
        u32 c=*p++;
        if (c<0x80) { *out++=(char16_t)c; continue; }
        if ((c&0xE0)==0xC0 && *p) c=((c&0x1F)<<6)|(*p++&0x3F);
        else if ((c&0xF0)==0xE0 && p[0] && p[1]) { c=((c&0x0F)<<12)|((p[0]&0x3F)<<6)|(p[1]&0x3F); p+=2; }
        else { c='?'; }
        if (c<=0xFFFF) *out++=(char16_t)c;
    }
    *out=0;
}

bool RetroLibraryMetadata::Load(const char* fileName)
{
    char path[384];
    snprintf(path,sizeof(path),"/_pico/metadata/user/%s.json",fileName);
    File f;
    if (f.Open(path, FA_READ|FA_OPEN_EXISTING)!=FR_OK) return false;
    u32 sz=f.GetSize(); if (!sz || sz>8192) return false;
    auto data=std::unique_ptr<u8[]>(new(cache_align) u8[sz+1]);
    u32 got=0; if (f.Read(data.get(),sz,got)!=FR_OK) return false; data[got]=0;
    DynamicJsonDocument json(8192);
    if (deserializeJson(json,data.get(),got)!=DeserializationError::Ok) return false;
    const char* title=json["title"] | "";
    const char* desc=json["shortDescriptionUa"].as<const char*>();
    if (!desc || !desc[0]) desc=json["shortDescription"] | "";
    if (!title[0] && !desc[0]) return false;
    char16_t* o=_display; char16_t* end=_display+767;
    appendUtf8(o,end,title); if (o<end) *o++='\n'; if (o<end) *o++='\n'; *o=0;
    appendUtf8(o,end,desc);
    return true;
}
