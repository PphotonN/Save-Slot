{

    _saveSettingsPending = true;
    _ioTaskQueue->Enqueue([this] (const vu8&)
    {

        _saveSettingsPending = false;

    });
}

void RomBrowserController::CycleBacklight()
{
    int level = GetBacklightLevel();
    if (level < 0) level = 0;
    SetBacklightLevel((level + 1) & 3);
}

void RomBrowserController::SetShowHints(bool value)

    _appSettingsService->GetAppSettings().showHints = value;
    SaveUiSettingsAsync();
    _stateMachine.Fire(RomBrowserStateTrigger::ChangeDisplayMode);
}

void RomBrowserController::CycleThemeColor()
{
    auto& s = _appSettingsService->GetAppSettings();
    s.themeColor = (s.themeColor + 1) % 6;
    ConsoleUiState::SetThemeColor(s.themeColor);
    SaveUiSettingsAsync();
}

void RomBrowserController::CycleLibraryTab(int delta)
{
    auto& s = _appSettingsService->GetAppSettings();
    int tab = (int)s.libraryTab + delta;
    while (tab < 0) tab += 5;
    while (tab >= 5) tab -= 5;
    s.libraryTab = (u8)tab;
    _favoritesFilter = (tab == 0);
    _completedFilter = false;
    SaveUiSettingsAsync();
    NavigateToPath(GetLibraryPath(s.libraryTab));
}
